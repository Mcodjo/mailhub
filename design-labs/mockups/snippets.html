            'html'    => $request->input('html'),

}

import { MailHubClient } from "@mailpro-hub/sdk";

const client = new MailHubClient({
  apiKey: process.env.MAILHUB_API_KEY, // jamais en dur dans le code
});

export async function sendProMail(to: string, subject: string, html: string) {
  return client.send({
    from: "contact@agence-nova.fr", // envoi au nom de l'adresse pro (Gmail OAuth)
    to, subject, html,
  });
}
// wp-config.php (ajouter une constante — valeur factice)
define( 'MAILHUB_API_KEY', 'mph_demo_sk_0000FAKE0000' );

// functions.php — formulaire de contact
function mailhub_send_contact( $to, $subject, $html ) {
    $response = wp_remote_post( 'https://api.mailpro-hub.demo/v1/send', array(
        'headers' => array(
            'Authorization' => 'Bearer ' . MAILHUB_API_KEY,
            'Content-Type'  => 'application/json',
        ),
        'body' => wp_json_encode( array(
            'from'    => 'contact@agence-nova.fr',
            'to'      => $to,
            'subject' => $subject,
            'html'    => $html,
        ) ),
    ) );
    return 200 === wp_remote_retrieve_response_code( $response );
}
